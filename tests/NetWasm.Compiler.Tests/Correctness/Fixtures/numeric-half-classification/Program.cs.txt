    private static int _bits;
    private static int _operation;
    private static int _actual;

    public static int Run(int input)
    {
        _bits = Bits[input / 12];
        _operation = input % 12;
        var value = BitConverter.UInt16BitsToHalf((ushort)_bits);
        var exponent = (_bits >> 10) & 31;
        var fraction = _bits & 1023;
        var negative = (_bits & 32768) != 0;
        var nan = exponent == 31 && fraction != 0;
        var infinity = exponent == 31 && fraction == 0;
        var zero = exponent == 0 && fraction == 0;
        var expected = _operation switch
        {
            0 => exponent != 31 ? 1 : 0,
            1 => infinity ? 1 : 0,
            2 => nan ? 1 : 0,
            3 => exponent != 0 && exponent != 31 ? 1 : 0,
            4 => exponent == 0 && fraction != 0 ? 1 : 0,
            5 => negative ? 1 : 0,
            6 => !negative ? 1 : 0,
            7 => infinity && !negative ? 1 : 0,
            8 => infinity && negative ? 1 : 0,
            9 or 10 => zero ? 1 : 0,
            _ => nan ? -1 : zero ? 0 : negative ? -1 : 1,
        };
        _actual = _operation switch
        {
            0 => Half.IsFinite(value) ? 1 : 0,
            1 => Half.IsInfinity(value) ? 1 : 0,
            2 => Half.IsNaN(value) ? 1 : 0,
            3 => Half.IsNormal(value) ? 1 : 0,
            4 => Half.IsSubnormal(value) ? 1 : 0,
            5 => Half.IsNegative(value) ? 1 : 0,
            6 => Half.IsPositive(value) ? 1 : 0,
            7 => Half.IsPositiveInfinity(value) ? 1 : 0,
            8 => Half.IsNegativeInfinity(value) ? 1 : 0,
            9 => value == Half.Zero ? 1 : 0,
            10 => value.Equals(Half.Zero) ? 1 : 0,
            _ => Normalize(value.CompareTo(Half.Zero)),
        };
        return _actual == expected ? 42 : -1;
    }

    private static int Normalize(int value) => value < 0 ? -1 : value > 0 ? 1 : 0;
    public static int Trace() => 3;
    public static int TraceCount() => 3;
    public static int TraceKind(int index) => 2;
    public static int TraceEventId(int index) => index;
    public static int TracePayloadLow(int index) => index == 0 ? _bits : index == 1 ? _operation : _actual;
    public static int TracePayloadHigh(int index) => 0;
}
