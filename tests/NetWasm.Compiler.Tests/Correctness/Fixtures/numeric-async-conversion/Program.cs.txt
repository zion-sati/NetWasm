    public static int Run(int input)
    {
        var index = input / 4;
        var route = input % 4;
        var synchronous = (route & 1) == 0;
        var first = new ControlledAwaitTransport.Gate(synchronous);
        var second = new ControlledAwaitTransport.Gate(synchronous);
        var progress = new ControlledAwaitTransport.Progress();
        var task = (Kinds[index], route < 2) switch
        {
            (0, true) => Int32Task(index, first, second, progress),
            (0, false) => Int32ValueTask(index, first, second, progress).AsTask(),
            (1, true) => UInt32Task(index, first, second, progress),
            (1, false) => UInt32ValueTask(index, first, second, progress).AsTask(),
            (2, true) => Int64Task(index, first, second, progress),
            (2, false) => Int64ValueTask(index, first, second, progress).AsTask(),
            (3, true) => UInt64Task(index, first, second, progress),
            (3, false) => UInt64ValueTask(index, first, second, progress).AsTask(),
            _ => throw new ArgumentOutOfRangeException(nameof(input)),
        };
        return ControlledAwaitTransport.CompleteOperation(task, first, second, progress, synchronous);
    }

    private static async Task<int> Int32Task(
        int index, ControlledAwaitTransport.Gate first, ControlledAwaitTransport.Gate second,
        ControlledAwaitTransport.Progress progress)
    {
        var source = unchecked((int)Sources[index]);
        progress.Step = 1;
        await first;
        if (unchecked((ulong)source) != Sources[index]) return -1;
        if (Widths[index] == 32)
        {
            var converted = (float)source;
            if (BitConverter.SingleToUInt32Bits(converted) != ExpectedFloating[index]) return -2;
            progress.Step = 2;
            await second;
            if (BitConverter.SingleToUInt32Bits(converted) != ExpectedFloating[index]) return -3;
            progress.Step = 3;
            var returned = checked((int)converted);
            return unchecked((ulong)returned) == ExpectedBack[index] ? 42 : -4;
        }
        else
        {
            var converted = (double)source;
            if (BitConverter.DoubleToUInt64Bits(converted) != ExpectedFloating[index]) return -2;
            progress.Step = 2;
            await second;
            if (BitConverter.DoubleToUInt64Bits(converted) != ExpectedFloating[index]) return -3;
            progress.Step = 3;
            var returned = checked((int)converted);
            return unchecked((ulong)returned) == ExpectedBack[index] ? 42 : -4;
        }
    }
