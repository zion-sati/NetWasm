    private static ulong _source;

    public static int Run(int input)
    {
        var row = input / 2;
        _source = Sources[row];
        _result = 0;
        Half result;
        if (Widths[row] == 32)
        {
            var value = BitConverter.UInt32BitsToSingle((uint)_source);
            result = (input & 1) == 0 ? unchecked((Half)value) : checked((Half)value);
        }
        else
        {
            var value = BitConverter.UInt64BitsToDouble(_source);
            result = (input & 1) == 0 ? unchecked((Half)value) : checked((Half)value);
        }
        if (Expected[row] == -1)
        {
            _result = Half.IsNaN(result) ? 1 : 0;
            return _result == 1 ? 42 : -1;
        }
        _result = BitConverter.HalfToUInt16Bits(result);
        return _result == Expected[row] ? 42 : -2;
    }

    public static int Trace() => 2;

    public static int TraceEventId(int index) => index;
    public static int TracePayloadLow(int index) => index == 0 ? unchecked((int)_source) : _result;
    public static int TracePayloadHigh(int index) => index == 0 ? unchecked((int)(_source >> 32)) : 0;
