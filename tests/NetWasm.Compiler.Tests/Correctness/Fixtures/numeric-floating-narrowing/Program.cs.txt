    private static long _source;
    private static long _narrowed;
    private static long _checked;
    private static long _widened;
    private static bool _nan;

    public static int Run(int input)
    {
        _source = Sources[input];
        var source = BitConverter.Int64BitsToDouble(_source);
        var narrowed = (float)source;
        var checkedNarrowed = checked((float)source);
        var widened = (double)narrowed;
        _nan = (unchecked((ulong)_source) & 0x7fffffffffffffffUL) > 0x7ff0000000000000UL;
        if (_nan)
        {
            _narrowed = float.IsNaN(narrowed) ? 1 : 0;
            _checked = float.IsNaN(checkedNarrowed) ? 1 : 0;
            _widened = double.IsNaN(widened) ? 1 : 0;
            if (_narrowed != 1) return -1;
            if (_checked != 1) return -2;
            if (_widened != 1) return -3;
        }
        else
        {
            _narrowed = BitConverter.SingleToInt32Bits(narrowed);
            _checked = BitConverter.SingleToInt32Bits(checkedNarrowed);
            _widened = BitConverter.DoubleToInt64Bits(widened);
            if (_narrowed != ExpectedSingle[input]) return -1;
            if (_checked != ExpectedSingle[input]) return -2;
            if (_widened != ExpectedWidened[input]) return -3;
        }
        return 42;
    }

    public static int Trace() => 4;
    public static int TraceCount() => 4;
    public static int TraceKind(int index) => index == 0 ? 5 : _nan ? 1 : index == 3 ? 5 : 4;
    public static int TraceEventId(int index) => index;
    private static long Payload(int index) => index switch { 0 => _source, 1 => _narrowed, 2 => _checked, 3 => _widened, _ => throw new ArgumentOutOfRangeException(nameof(index)) };
    public static int TracePayloadLow(int index) => unchecked((int)Payload(index));
    public static int TracePayloadHigh(int index) => unchecked((int)(Payload(index) >> 32));
