    public static int Run(int input) => Single[input] ? RunSingle(input) : RunDouble(input);

    private static int RunSingle(int input)
    {
        var value = BitConverter.UInt32BitsToSingle((uint)Inputs[input]);
        var result = Operations[input] switch
        {
            0 => MathF.Sqrt(value),
            1 => MathF.Cbrt(value),
            2 => MathF.Sin(value),
            3 => MathF.Cos(value),
            4 => MathF.Tan(value),
            5 => MathF.Asin(value),
            6 => MathF.Acos(value),
            7 => MathF.Atan(value),
            8 => MathF.Exp(value),
            9 => MathF.Log(value),
            10 => MathF.Log2(value),
            11 => MathF.Log10(value),
            12 => MathF.Sinh(value),
            13 => MathF.Cosh(value),
            14 => MathF.Tanh(value),
            15 => MathF.Asinh(value),
            16 => MathF.Acosh(value),
            17 => MathF.Atanh(value),
            _ => throw new ArgumentOutOfRangeException(nameof(input)),
        };
        if (ExpectedNaN[input]) return float.IsNaN(result) ? 42 : -1;
        return BitConverter.SingleToUInt32Bits(result) == (uint)Expected[input] ? 42 : -2;
    }

    private static int RunDouble(int input)
    {

            17 => Math.Atanh(value),
            _ => throw new ArgumentOutOfRangeException(nameof(input)),
        };
        if (ExpectedNaN[input]) return double.IsNaN(result) ? 42 : -1;
        return BitConverter.DoubleToUInt64Bits(result) == Expected[input] ? 42 : -2;
    }
