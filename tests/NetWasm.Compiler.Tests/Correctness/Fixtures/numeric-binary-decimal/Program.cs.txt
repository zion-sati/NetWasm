    public static int Run(int input)
    {
        var row = input / 4;
        var operation = input % 4;
        decimal result;
        if (Single[row])
        {
            var value = BitConverter.UInt32BitsToSingle((uint)Bits[row]);
            result = operation switch
            {
                0 => unchecked((decimal)value),
                1 => checked((decimal)value),
                2 => new decimal(value),
                _ => Convert.ToDecimal(value),
            };
        }
        else
        {
            var value = BitConverter.UInt64BitsToDouble(Bits[row]);
            result = operation switch
            {
                0 => unchecked((decimal)value),
                1 => checked((decimal)value),
                2 => new decimal(value),
                _ => Convert.ToDecimal(value),
            };
        }
        return MatchesExpected(row, result) ? 42 : -1;
    }

    private static bool MatchesExpected(int row, decimal result)
    {
        var actual = decimal.GetBits(result);
        var start = row * 4;
        return unchecked((uint)actual[0]) == ExpectedWords[start] &&
            unchecked((uint)actual[1]) == ExpectedWords[start + 1] &&
            unchecked((uint)actual[2]) == ExpectedWords[start + 2] &&
            unchecked((uint)actual[3]) == ExpectedWords[start + 3];
    }
