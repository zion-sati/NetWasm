    public static int Run(int input)
    {
        for (var index = 0; index < Observations.Length; index++) Observations[index] = 0;
        return input < 348 ? Representation(input) : Rounding(input - 348);
    }

    private static int Representation(int input)
    {
        var part = input % 6;
        var scale = (byte)(input / 12);
        var negative = (input / 6 & 1) != 0;
        var expectedFlags = (scale << 16) | (negative ? int.MinValue : 0);
        var value = new decimal(Low[part], Middle[part], High[part], negative, scale);
        var bits = decimal.GetBits(value);
        if (bits.Length != 4) return -1;
        for (var index = 0; index < 4; index++) Observations[index] = bits[index];
        if (bits[0] != Low[part] || bits[1] != Middle[part] ||
            bits[2] != High[part] || bits[3] != expectedFlags) return -2;
        var restored = new decimal(bits);
        var restoredBits = decimal.GetBits(restored);
        for (var index = 0; index < 4; index++)
            if (restoredBits[index] != bits[index]) return -3;
        if (value != restored || !value.Equals(restored) || value.CompareTo(restored) != 0) return -4;
        if (value.GetHashCode() != restored.GetHashCode()) return -5;
        if (part == 0)
        {
            if (value != decimal.Zero || !value.Equals(decimal.Zero) || value.CompareTo(decimal.Zero) != 0) return -6;
            if (value.GetHashCode() != decimal.Zero.GetHashCode()) return -7;
        }
        return 42;
    }

    private static int Rounding(int input)
    {
        var row = input / 5;
        var mode = (MidpointRounding)(input % 5);
        var coefficient = Coefficients[row];
        var value = new decimal(coefficient < 0 ? -coefficient : coefficient, 0, 0, coefficient < 0, Scales[row]);
        var expected = Expected[input];
        var rounded = decimal.Round(value, Digits[row], mode);
        Observations[0] = rounded == expected ? 1 : 0;
        if (Observations[0] == 0) return -8;
        Observations[1] = Math.Round(value, Digits[row], mode) == expected ? 1 : 0;
        if (Observations[1] == 0) return -9;
        if (mode == MidpointRounding.ToEven)
        {
            Observations[2] = decimal.Round(value, Digits[row]) == expected ? 1 : 0;
            if (Observations[2] == 0) return -10;
        }
        if (Digits[row] == 0)
        {
            var integral = mode switch
            {
                MidpointRounding.ToZero => decimal.Truncate(value),
                MidpointRounding.ToNegativeInfinity => decimal.Floor(value),
                MidpointRounding.ToPositiveInfinity => decimal.Ceiling(value),
                _ => decimal.Round(value, mode),
            };
            Observations[3] = integral == expected ? 1 : 0;
            if (Observations[3] == 0) return -11;
        }
        return 42;
    }
