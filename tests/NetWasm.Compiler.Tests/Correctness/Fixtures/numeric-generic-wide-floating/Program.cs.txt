    public static int Run(int input)
    {
        var index = input / 6;
        var operation = input % 6;
        if (Signed[index])
        {
            var value = new Int128(High[index], Low[index]);
            if (operation < 3)
            {
                var actual = Convert<Int128, float>(value, operation);
                return BitConverter.SingleToUInt32Bits(actual) == (uint)SingleExpected[index] ? 42 : -1;
            }
            else
            {
                var actual = Convert<Int128, double>(value, operation - 3);
                return BitConverter.DoubleToUInt64Bits(actual) == DoubleExpected[index] ? 42 : -2;
            }
        }
        else
        {
            var value = new UInt128(High[index], Low[index]);
            if (operation < 3)
            {
                var actual = Convert<UInt128, float>(value, operation);
                return BitConverter.SingleToUInt32Bits(actual) == (uint)SingleExpected[index] ? 42 : -1;
            }
            else
            {
                var actual = Convert<UInt128, double>(value, operation - 3);
                return BitConverter.DoubleToUInt64Bits(actual) == DoubleExpected[index] ? 42 : -2;
            }
        }
    }
