    private static long _operand;
    private static long _checked;
    private static long _unchecked;
    private static int _count;
    private static int _floatingKind;

    public static int Run(int input)
    {
        var single = input < 88;
        var destination = input < 176 ? (input / 22) % 4 : 0;
        var bits = OperandBits[input];
        _operand = single ? unchecked((int)bits) : unchecked((long)bits);
        _floatingKind = single ? 4 : 5;
        _checked = _unchecked = 0;
        _count = 1;
        if (single)
        {
            var source = BitConverter.Int32BitsToSingle(unchecked((int)bits));
            _checked = destination switch
            {
                0 => checked((int)source),
                1 => checked((uint)source),
                2 => checked((long)source),
                3 => unchecked((long)checked((ulong)source)),
                _ => throw new ArgumentOutOfRangeException(nameof(input)),
            };
            _count = 2;
            _unchecked = destination switch
            {
                0 => unchecked((int)source),
                1 => unchecked((uint)source),
                2 => unchecked((long)source),
                3 => unchecked((long)(ulong)source),
                _ => throw new ArgumentOutOfRangeException(nameof(input)),
            };
        }
        else
        {
            var source = BitConverter.Int64BitsToDouble(unchecked((long)bits));
            _checked = destination switch
            {
                0 => checked((int)source),
                1 => checked((uint)source),
                2 => checked((long)source),
                3 => unchecked((long)checked((ulong)source)),
                _ => throw new ArgumentOutOfRangeException(nameof(input)),
            };
            _count = 2;
            _unchecked = destination switch
            {
                0 => unchecked((int)source),
                1 => unchecked((uint)source),
                2 => unchecked((long)source),
                3 => unchecked((long)(ulong)source),
                _ => throw new ArgumentOutOfRangeException(nameof(input)),
            };
        }
        _count = 3;
        if (_checked != ExpectedBits[input]) return -1;
        if (_unchecked != ExpectedBits[input]) return -2;
        return 42;
    }

    public static int Trace() => _count;
    public static int TraceCount() => _count;
    public static int TraceKind(int index) => index == 0 ? _floatingKind : 3;
    public static int TraceEventId(int index) => index;
    private static long Payload(int index) => index switch { 0 => _operand, 1 => _checked, 2 => _unchecked, _ => throw new ArgumentOutOfRangeException(nameof(index)) };
    public static int TracePayloadLow(int index) => unchecked((int)Payload(index));
    public static int TracePayloadHigh(int index) => unchecked((int)(Payload(index) >> 32));
}
