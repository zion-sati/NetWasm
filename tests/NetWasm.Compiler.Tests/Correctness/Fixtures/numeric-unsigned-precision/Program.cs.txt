    private static long _source;
    private static long _floating;
    private static long _returned;
    private static int _kind;
    private static int _count;

    public static int Run(int input)
    {
        var single = input < 10 || (input >= 20 && input < 44);
        var wide = input >= 20;
        var source = Sources[input];
        _source = unchecked((long)source);
        _floating = _returned = 0;
        _kind = single ? 4 : 5;
        _count = 1;
        if (single)
        {
            // Keep distinct UInt32 and UInt64 conversions in the emitted CIL.
            var converted = wide ? (float)source : (float)(uint)source;
            _floating = BitConverter.SingleToInt32Bits(converted);
            _count = 2;
            if (_floating != FloatingBits[input]) return -1;
            var returned = wide ? checked((ulong)converted) : checked((uint)converted);
            _returned = unchecked((long)returned);
            _count = 3;
            if (returned != ReturnedIntegers[input]) return -2;
        }
        else
        {
            var converted = wide ? (double)source : (double)(uint)source;
            _floating = BitConverter.DoubleToInt64Bits(converted);
            _count = 2;
            if (_floating != FloatingBits[input]) return -1;
            var returned = wide ? checked((ulong)converted) : checked((uint)converted);
            _returned = unchecked((long)returned);
            _count = 3;
            if (returned != ReturnedIntegers[input]) return -2;
        }
        return 42;
    }

    public static int Trace() => _count;
    public static int TraceCount() => _count;
    public static int TraceKind(int index) => index == 1 ? _kind : 3;
    public static int TraceEventId(int index) => index;
    private static long Payload(int index) => index switch { 0 => _source, 1 => _floating, 2 => _returned, _ => throw new ArgumentOutOfRangeException(nameof(index)) };
    public static int TracePayloadLow(int index) => unchecked((int)Payload(index));
    public static int TracePayloadHigh(int index) => unchecked((int)(Payload(index) >> 32));
}
