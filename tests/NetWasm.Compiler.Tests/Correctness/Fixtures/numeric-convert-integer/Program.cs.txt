    public static int Run(int input)
    {
        var actual = Widths[input] == 32
            ? ConvertSingle(BitConverter.UInt32BitsToSingle((uint)Sources[input]), Destinations[input])
            : ConvertDouble(BitConverter.UInt64BitsToDouble(Sources[input]), Destinations[input]);
        return actual == Expected[input] ? 42 : -1;
    }

    private static ulong ConvertSingle(float value, int destination) =>
        destination switch
        {
            0 => unchecked((ulong)Convert.ToSByte(value)),
            1 => unchecked((ulong)Convert.ToByte(value)),
            2 => unchecked((ulong)Convert.ToInt16(value)),
            3 => unchecked((ulong)Convert.ToUInt16(value)),
            4 => unchecked((ulong)Convert.ToInt32(value)),
            5 => unchecked((ulong)Convert.ToUInt32(value)),
            6 => unchecked((ulong)Convert.ToInt64(value)),
            7 => unchecked((ulong)Convert.ToUInt64(value)),
            _ => throw new ArgumentOutOfRangeException(nameof(destination)),
        };

    private static ulong ConvertDouble(double value, int destination) =>
        destination switch
        {
            0 => unchecked((ulong)Convert.ToSByte(value)),
            1 => unchecked((ulong)Convert.ToByte(value)),
            2 => unchecked((ulong)Convert.ToInt16(value)),
            3 => unchecked((ulong)Convert.ToUInt16(value)),
            4 => unchecked((ulong)Convert.ToInt32(value)),
            5 => unchecked((ulong)Convert.ToUInt32(value)),
            6 => unchecked((ulong)Convert.ToInt64(value)),
            7 => unchecked((ulong)Convert.ToUInt64(value)),
            _ => throw new ArgumentOutOfRangeException(nameof(destination)),
        };
