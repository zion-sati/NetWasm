        return result == ExpectedInt128_UInt128[input] ? 42 : -1;
    }

    private static int CastUInt128_long(int input)
    {
        var value = ValuesUInt128[input / 5];
        var result = (input % 5) switch
        {
            0 => unchecked((long)value),
            1 => checked((long)value),
            2 => Checked<UInt128, long>(value),
            3 => Saturating<UInt128, long>(value),
            _ => Truncating<UInt128, long>(value),
        };
        Observe(result);
        return result == ExpectedUInt128_long[input] ? 42 : -1;
    }

    private static TTarget Checked<TSource, TTarget>(TSource value)
        where TSource : INumberBase<TSource>
        where TTarget : INumberBase<TTarget> => TTarget.CreateChecked(value);

    private static TTarget Saturating<TSource, TTarget>(TSource value)
        where TSource : INumberBase<TSource>
        where TTarget : INumberBase<TTarget> => TTarget.CreateSaturating(value);

    private static TTarget Truncating<TSource, TTarget>(TSource value)
        where TSource : INumberBase<TSource>
        where TTarget : INumberBase<TTarget> => TTarget.CreateTruncating(value);

    private static void Observe(long value)
    {
        _low = unchecked((ulong)value);
        _high = value < 0 ? ulong.MaxValue : 0;
    }

    private static void Observe(ulong value)
    {
        _low = value;
        _high = 0;
    }

    private static void Observe(Int128 value)
    {
        _low = unchecked((ulong)value);
        _high = unchecked((ulong)(value >> 64));
    }

    private static void Observe(UInt128 value)
    {
        _low = unchecked((ulong)value);
        _high = unchecked((ulong)(value >> 64));
    }

    public static int Trace() => 2;
    public static int TraceCount() => 2;
    public static int TraceKind(int index) => 3;
    public static int TraceEventId(int index) => index;
    public static int TracePayloadLow(int index) => unchecked((int)(index == 0 ? _low : _high));
    public static int TracePayloadHigh(int index) => unchecked((int)((index == 0 ? _low : _high) >> 32));
