    public static int Run(int input)
    {
        if (input >= 648) return PowSpecialValueSign(input - 648);
        if (Single[input])
        {
            var left = BitConverter.UInt32BitsToSingle((uint)Left[input]);
            var right = BitConverter.UInt32BitsToSingle((uint)Right[input]);
            var result = Operations[input] == 0 ? MathF.Pow(left, right) : MathF.Atan2(left, right);
            if (ExpectedNaN[input]) return float.IsNaN(result) ? 42 : -1;
            if (float.IsNaN(result)) return -2;
            var actual = BitConverter.SingleToUInt32Bits(result);
            var expected = (uint)Expected[input];
            if (Tolerances[input] == 0) return actual == expected ? 42 : -3;
            if ((actual & 0x80000000U) != (expected & 0x80000000U)) return -4;
            var distance = actual >= expected ? actual - expected : expected - actual;
            return distance <= Tolerances[input] ? 42 : -5;
        }
        else
        {
            var left = BitConverter.UInt64BitsToDouble(Left[input]);
            var right = BitConverter.UInt64BitsToDouble(Right[input]);
            var result = Operations[input] == 0 ? Math.Pow(left, right) : Math.Atan2(left, right);
            if (ExpectedNaN[input]) return double.IsNaN(result) ? 42 : -1;
            if (double.IsNaN(result)) return -2;
            var actual = BitConverter.DoubleToUInt64Bits(result);
            var expected = Expected[input];
            if (Tolerances[input] == 0) return actual == expected ? 42 : -3;
            if ((actual & 0x8000000000000000UL) != (expected & 0x8000000000000000UL)) return -4;
            var distance = actual >= expected ? actual - expected : expected - actual;
            return distance <= Tolerances[input] ? 42 : -5;
        }
    }

    private static int PowSpecialValueSign(int input)
    {
        // Both exponent signs, fractional neighbours of one, precision boundaries,
        // and extreme finite exponents. Only pairs 0, 2 and 8 are odd integers.
        uint[] singleExponents =
        [
            0x3f800000U, 0x40000000U, 0x40400000U, 0x3f000000U,
            0x3fc00000U, 0x40600000U, 0x3f7fffffU, 0x3f800001U,
            0x4b7fffffU, 0x4b800000U, 0x7f7fffffU, 0x00000001U,
        ];
        ulong[] doubleExponents =
        [
            0x3ff0000000000000UL, 0x4000000000000000UL,
            0x4008000000000000UL, 0x3fe0000000000000UL,
            0x3ff8000000000000UL, 0x400c000000000000UL,
            0x3fefffffffffffffUL, 0x3ff0000000000001UL,
            0x433fffffffffffffUL, 0x4340000000000000UL,
            0x7fefffffffffffffUL, 0x0000000000000001UL,
        ];
        var single = input < 96;
        var basis = input / 24 % 4;
        var pair = input % 24 / 2;
        var negativeExponent = (input & 1) != 0;
        var negativeBase = (basis & 1) != 0;
        var infiniteBase = basis >= 2;
        var negativeResult = negativeBase && pair is 0 or 2 or 8;
        var infiniteResult = infiniteBase != negativeExponent;
        if (single)
        {
            var basisBits = (infiniteBase ? 0x7f800000U : 0U) |
                (negativeBase ? 0x80000000U : 0U);
            var exponentBits = singleExponents[pair] |
                (negativeExponent ? 0x80000000U : 0U);
            var expected = (infiniteResult ? 0x7f800000U : 0U) |
                (negativeResult ? 0x80000000U : 0U);
            var result = MathF.Pow(BitConverter.UInt32BitsToSingle(basisBits),
                BitConverter.UInt32BitsToSingle(exponentBits));
            return BitConverter.SingleToUInt32Bits(result) == expected ? 42 : -6;
        }
        else
        {
            var basisBits = (infiniteBase ? 0x7ff0000000000000UL : 0UL) |
                (negativeBase ? 0x8000000000000000UL : 0UL);
            var exponentBits = doubleExponents[pair] |
                (negativeExponent ? 0x8000000000000000UL : 0UL);
            var expected = (infiniteResult ? 0x7ff0000000000000UL : 0UL) |
                (negativeResult ? 0x8000000000000000UL : 0UL);
            var result = Math.Pow(BitConverter.UInt64BitsToDouble(basisBits),
                BitConverter.UInt64BitsToDouble(exponentBits));
            return BitConverter.DoubleToUInt64Bits(result) == expected ? 42 : -6;
        }
    }
