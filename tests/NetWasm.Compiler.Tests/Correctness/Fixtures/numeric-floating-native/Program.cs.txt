    public static int Run(int input)
    {
        if (input >= Inputs.Length) return RunTransportControl(input - Inputs.Length);
        var narrow = sizeof(nint) == 4;
        var expectedOverflow = narrow ? Overflow32[input] : Overflow64[input];
        var expected = narrow ? Expected32[input] : Expected64[input];
        try
        {
            ulong actual;
            if (Single[input])
            {
                var value = BitConverter.UInt32BitsToSingle((uint)Inputs[input]);
                if (Operations[input] < 2)
                {
                    var result = Operations[input] == 0 ? unchecked((nint)value) : checked((nint)value);
                    actual = narrow ? unchecked((uint)result) : unchecked((ulong)result);
                }
                else
                {
                    var result = Operations[input] == 2 ? unchecked((nuint)value) : checked((nuint)value);
                    actual = narrow ? unchecked((uint)result) : unchecked((ulong)result);
                }
            }
            else
            {
                var value = BitConverter.UInt64BitsToDouble(Inputs[input]);
                if (Operations[input] < 2)
                {
                    var result = Operations[input] == 0 ? unchecked((nint)value) : checked((nint)value);
                    actual = narrow ? unchecked((uint)result) : unchecked((ulong)result);
                }
                else
                {
                    var result = Operations[input] == 2 ? unchecked((nuint)value) : checked((nuint)value);
                    actual = narrow ? unchecked((uint)result) : unchecked((ulong)result);
                }
            }
            if (expectedOverflow) return -1;
            return actual == expected ? 42 : -3;
        }
        catch (OverflowException)
        {
            return expectedOverflow ? 42 : -2;
        }
    }

    // Additional ordinary Roslyn controls preserve the floating source across a
    // call and the native result across array storage, for both signednesses,
    // precisions and overflow policies. Negative fractions truncate to zero.
    private static int RunTransportControl(int input)
    {
        var value = (input & 1) == 0 ? -0.75 : 42.75;
        var expected = (input & 1) == 0 ? 0 : 42;
        var check = (input & 4) != 0;
        if ((input & 2) == 0)
        {
            var values = new nint[1];
            values[0] = (input & 8) == 0
                ? ConvertSigned((float)value, check)
                : ConvertSigned(value, check);
            return values[0] == expected && value == ((input & 1) == 0 ? -0.75 : 42.75) ? 42 : -4;
        }
        else
        {
            var values = new nuint[1];
            values[0] = (input & 8) == 0
                ? ConvertUnsigned((float)value, check)
                : ConvertUnsigned(value, check);
            return values[0] == (nuint)expected && value == ((input & 1) == 0 ? -0.75 : 42.75) ? 42 : -5;
        }
    }

    private static nint ConvertSigned(float value, bool check) =>
        check ? checked((nint)value) : unchecked((nint)value);

    private static nint ConvertSigned(double value, bool check) =>
        check ? checked((nint)value) : unchecked((nint)value);

    private static nuint ConvertUnsigned(float value, bool check) =>
        check ? checked((nuint)value) : unchecked((nuint)value);

    private static nuint ConvertUnsigned(double value, bool check) =>
        check ? checked((nuint)value) : unchecked((nuint)value);
