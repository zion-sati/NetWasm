    private static long _left;
    private static long _right;
    private static int _operators;
    private static int _branches;
    private static int _equals;
    private static int _order;
    private static int _kind;

    public static int Run(int input)
    {
        var single = input < 196;
        var pair = input % 196;
        var left = pair / 14;
        var right = pair % 14;
        _kind = single ? 4 : 5;
        if (single)
        {
            _left = Singles[left];
            _right = Singles[right];
            Observe(BitConverter.Int32BitsToSingle(Singles[left]), BitConverter.Int32BitsToSingle(Singles[right]));
        }
        else
        {
            _left = Doubles[left];
            _right = Doubles[right];
            Observe(BitConverter.Int64BitsToDouble(Doubles[left]), BitConverter.Int64BitsToDouble(Doubles[right]));
        }
        if (_operators != ExpectedOperators[pair]) return -1;
        if (_branches != (ExpectedOperators[pair] ^ 63)) return -2;
        if (_equals != ExpectedEquals[pair]) return -3;
        if (_order != ExpectedOrder[pair]) return -4;
        return 42;
    }

    private static void Observe(float left, float right)
    {
        _operators = (left == right ? 1 : 0) | (left != right ? 2 : 0)
            | (left < right ? 4 : 0) | (left > right ? 8 : 0)
            | (left <= right ? 16 : 0) | (left >= right ? 32 : 0);
        _branches = 0;
        if (!(left == right)) _branches |= 1;
        if (!(left != right)) _branches |= 2;
        if (!(left < right)) _branches |= 4;
        if (!(left > right)) _branches |= 8;
        if (!(left <= right)) _branches |= 16;
        if (!(left >= right)) _branches |= 32;
        _equals = left.Equals(right) ? 1 : 0;
        var order = left.CompareTo(right);
        _order = order < 0 ? -1 : order > 0 ? 1 : 0;
    }

    private static void Observe(double left, double right)
    {
        _operators = (left == right ? 1 : 0) | (left != right ? 2 : 0)
            | (left < right ? 4 : 0) | (left > right ? 8 : 0)
            | (left <= right ? 16 : 0) | (left >= right ? 32 : 0);
        _branches = 0;
        if (!(left == right)) _branches |= 1;
        if (!(left != right)) _branches |= 2;
        if (!(left < right)) _branches |= 4;
        if (!(left > right)) _branches |= 8;
        if (!(left <= right)) _branches |= 16;
        if (!(left >= right)) _branches |= 32;
        _equals = left.Equals(right) ? 1 : 0;
        var order = left.CompareTo(right);
        _order = order < 0 ? -1 : order > 0 ? 1 : 0;
    }

    public static int Trace() => 6;
    public static int TraceCount() => 6;
    public static int TraceKind(int index) => index < 2 ? _kind : 2;
    public static int TraceEventId(int index) => index;
    private static long Payload(int index) => index switch { 0 => _left, 1 => _right, 2 => _operators, 3 => _branches, 4 => _equals, 5 => _order, _ => throw new ArgumentOutOfRangeException(nameof(index)) };
    public static int TracePayloadLow(int index) => unchecked((int)Payload(index));
    public static int TracePayloadHigh(int index) => unchecked((int)(Payload(index) >> 32));
}
