    private static long _left;
    private static long _right;
    private static long _result;
    private static int _operation;
    private static int _kind;
    private static bool _nan;

    public static int Run(int input)
    {
        if (input >= 160) return RunRemainder(input - 160);
        var single = input < 80;
        _operation = Operations[input];
        _kind = single ? 4 : 5;
        _left = single ? unchecked((int)LeftBits[input]) : unchecked((long)LeftBits[input]);
        _right = single ? unchecked((int)RightBits[input]) : unchecked((long)RightBits[input]);
        if (single)
        {
            var left = BitConverter.Int32BitsToSingle(unchecked((int)_left));
            var right = BitConverter.Int32BitsToSingle(unchecked((int)_right));
            var result = _operation switch
            {
                0 => left + right, 1 => left - right, 2 => left * right,
                3 => left / right, 4 => left % right, 5 => -left,
                _ => throw new ArgumentOutOfRangeException(nameof(input)),
            };
            _nan = (ExpectedBits[input] & 0x7fffffffUL) > 0x7f800000UL;
            _result = _nan ? (float.IsNaN(result) ? 1 : 0) : BitConverter.SingleToInt32Bits(result);
            if (_nan ? _result != 1 : _result != unchecked((int)ExpectedBits[input])) return -1;
        }
        else
        {
            var left = BitConverter.Int64BitsToDouble(_left);
            var right = BitConverter.Int64BitsToDouble(_right);
            var result = _operation switch
            {
                0 => left + right, 1 => left - right, 2 => left * right,
                3 => left / right, 4 => left % right, 5 => -left,
                _ => throw new ArgumentOutOfRangeException(nameof(input)),
            };
            _nan = (ExpectedBits[input] & 0x7fffffffffffffffUL) > 0x7ff0000000000000UL;
            _result = _nan ? (double.IsNaN(result) ? 1 : 0) : BitConverter.DoubleToInt64Bits(result);
            if (_nan ? _result != 1 : _result != unchecked((long)ExpectedBits[input])) return -1;
        }
        return 42;
    }

    public static int Trace() => 4;
    public static int TraceCount() => 4;
    public static int TraceKind(int index) => index == 3 ? 2 : index == 2 && _nan ? 1 : _kind;
    public static int TraceEventId(int index) => index;
    private static long Payload(int index) => index switch { 0 => _left, 1 => _right, 2 => _result, 3 => _operation, _ => throw new ArgumentOutOfRangeException(nameof(index)) };
    public static int TracePayloadLow(int index) => unchecked((int)Payload(index));
    public static int TracePayloadHigh(int index) => unchecked((int)(Payload(index) >> 32));
