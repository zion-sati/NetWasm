    private static int SignedOperation(int input)
    {
        var left = new Int128(LeftHigh[input], LeftLow[input]);
        var right = new Int128(RightHigh[input], RightLow[input]);
        var count = Counts[input];
        var result = Operations[input] switch
        {
            0 => unchecked(left + right),
            1 => checked(left + right),
            2 => unchecked(left - right),
            3 => checked(left - right),
            4 => unchecked(left * right),
            5 => checked(left * right),
            6 => left / right,
            7 => left % right,
            8 => left & right,
            9 => left | right,
            10 => left ^ right,
            11 => left << count,
            12 => left >> count,
            13 => left >>> count,
            14 => unchecked(-left),
            _ => checked(-left),
        };

        _high = unchecked((ulong)(result >> 64));
