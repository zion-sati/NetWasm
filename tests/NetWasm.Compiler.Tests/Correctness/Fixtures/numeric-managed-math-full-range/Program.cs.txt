    public static int Run(int input)
    {
        var x = BitConverter.UInt64BitsToDouble(Inputs[input]);
        var result = Operations[input] switch
        {
            0 => Math.Sin(x),
            1 => Math.Cos(x),
            2 => Math.Tan(x),
            3 => Math.Sinh(x),
            4 => Math.Tanh(x),
            5 => Math.Acosh(x),
            6 => Math.Exp(x),
            7 => Math.Log(x),
            _ => throw new ArgumentOutOfRangeException(nameof(input)),
        };
        var expected = Expected[input];
        var magnitude = expected & 0x7FFFFFFFFFFFFFFFUL;
        if (magnitude > 0x7FF0000000000000UL) return double.IsNaN(result) ? 42 : -1;
        var actual = BitConverter.DoubleToUInt64Bits(result);
        if (magnitude == 0 || magnitude == 0x7FF0000000000000UL) return actual == expected ? 42 : -2;
        if (!double.IsFinite(result) || (actual >> 63) != (expected >> 63)) return -3;
        var distance = actual >= expected ? actual - expected : expected - actual;
        return distance <= 4UL ? 42 : -4;
    }
