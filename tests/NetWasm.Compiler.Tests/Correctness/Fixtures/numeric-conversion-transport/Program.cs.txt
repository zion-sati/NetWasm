    public static int Run(int input)
    {
        var index = input / 7;
        var route = input % 7;
        return Kinds[index] switch
        {
            0 => Int32RoundTrip(index, route),
            1 => UInt32RoundTrip(index, route),
            2 => Int64RoundTrip(index, route),
            3 => UInt64RoundTrip(index, route),
            _ => throw new ArgumentOutOfRangeException(nameof(input)),
        };
    }

    private static int Int32RoundTrip(int index, int route)
    {
        var source = Transport(unchecked((int)Sources[index]), route);
        if (unchecked((ulong)source) != Sources[index]) return -1;
        if (Widths[index] == 32)
        {
            var converted = (float)source;
            if (BitConverter.SingleToUInt32Bits(converted) != ExpectedFloating[index]) return -2;
            var transported = Transport(converted, route);
            if (BitConverter.SingleToUInt32Bits(transported) != ExpectedFloating[index]) return -3;
            var returned = checked((int)transported);
            return unchecked((ulong)returned) == ExpectedBack[index] ? 42 : -4;
        }
        else
        {
            var converted = (double)source;
            if (BitConverter.DoubleToUInt64Bits(converted) != ExpectedFloating[index]) return -2;
            var transported = Transport(converted, route);
            if (BitConverter.DoubleToUInt64Bits(transported) != ExpectedFloating[index]) return -3;
            var returned = checked((int)transported);
            return unchecked((ulong)returned) == ExpectedBack[index] ? 42 : -4;
        }
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static T Transport<T>(T value, int route)
    {
        switch (route)
        {
            case 0: return Identity(value);
            case 1:
                var array = new T[2];
                array[1] = value;
                return array[1];
            case 2: return new Cell<T> { Value = value }.Value;
            case 3: return Select(false, default!, value);
            case 4: return Select(true, value, default!);
            case 5:
                Func<T, T> callback = Identity;
                return callback(value);
            case 6:
                Func<T> callbackWithCapture = () => value;
                return Invoke(callbackWithCapture);
            default: throw new ArgumentOutOfRangeException(nameof(route));
        }
    }

    private static T Identity<T>(T value) => value;

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static T Select<T>(bool first, T left, T right) => first ? left : right;

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static T Invoke<T>(Func<T> callback) => callback();

    private sealed class Cell<T>
    {
        public T Value = default!;
    }
}
