    public static int Run(int input)
    {
        var size = sizeof(nint);
        if (size != 4 && size != 8) return -5;
        if (IntPtr.Size != size || UIntPtr.Size != size) return -6;
        var wide = size == 8;
        if (input < 14) return Castsbyte_nint(input - 0, wide);
        if (input < 28) return Castsbyte_nuint(input - 14, wide);
        if (input < 42) return Castbyte_nint(input - 28, wide);
        if (input < 56) return Castbyte_nuint(input - 42, wide);
        if (input < 88) return Castshort_nint(input - 56, wide);
        if (input < 120) return Castshort_nuint(input - 88, wide);
        if (input < 146) return Castushort_nint(input - 120, wide);
        if (input < 172) return Castushort_nuint(input - 146, wide);
        if (input < 198) return Castchar_nint(input - 172, wide);
        if (input < 224) return Castchar_nuint(input - 198, wide);
        if (input < 274) return Castint_nint(input - 224, wide);
        if (input < 324) return Castint_nuint(input - 274, wide);
        if (input < 362) return Castuint_nint(input - 324, wide);
        if (input < 400) return Castuint_nuint(input - 362, wide);
        if (input < 468) return Castlong_nint(input - 400, wide);
        if (input < 536) return Castlong_nuint(input - 468, wide);
        if (input < 586) return Castulong_nint(input - 536, wide);
        if (input < 636) return Castulong_nuint(input - 586, wide);
        if (input < 704) return Castnint_sbyte(input - 636, wide);
        if (input < 772) return Castnint_byte(input - 704, wide);
        if (input < 840) return Castnint_short(input - 772, wide);
        if (input < 908) return Castnint_ushort(input - 840, wide);
        if (input < 976) return Castnint_char(input - 908, wide);
        if (input < 1044) return Castnint_int(input - 976, wide);
        if (input < 1112) return Castnint_uint(input - 1044, wide);
        if (input < 1180) return Castnint_long(input - 1112, wide);
        if (input < 1248) return Castnint_ulong(input - 1180, wide);
        if (input < 1316) return Castnint_nint(input - 1248, wide);
        if (input < 1384) return Castnint_nuint(input - 1316, wide);
        if (input < 1434) return Castnuint_sbyte(input - 1384, wide);
        if (input < 1484) return Castnuint_byte(input - 1434, wide);
        if (input < 1534) return Castnuint_short(input - 1484, wide);
        if (input < 1584) return Castnuint_ushort(input - 1534, wide);

        if (input < 1734) return Castnuint_uint(input - 1684, wide);
        if (input < 1784) return Castnuint_long(input - 1734, wide);
        if (input < 1834) return Castnuint_ulong(input - 1784, wide);
        if (input < 1884) return Castnuint_nint(input - 1834, wide);
        if (input < 1934) return Castnuint_nuint(input - 1884, wide);
        throw new ArgumentOutOfRangeException(nameof(input));
    }

    private static int Castsbyte_nint(int input, bool wide)
    {
        var index = input / 2;
        var value = Valuessbyte[index];
        var expected = wide ? Expected64sbyte_nint[index] : Expected32sbyte_nint[index];
        var shouldOverflow = (input & 1) != 0 && (wide ? Overflow64sbyte_nint[index] : Overflow32sbyte_nint[index]);
        nint result;
        try
        {
            result = (input & 1) != 0 ? checked((nint)value) : unchecked((nint)value);
        }
        catch (OverflowException)
        {
            return shouldOverflow ? 42 : -2;
        }
        if (shouldOverflow) return -3;
        return (long)result == expected ? 42 : -4;
    }
